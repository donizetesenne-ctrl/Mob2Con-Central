🚀 Quick Start

<!-- 2. Use as classes -->

  <div class="kpi-label">FATURAMENTO</div>
  <div class="kpi-value">R$ 2.4M</div>
  <div class="kpi-trend up">▲ 12.3%</div>
</div>

<!-- 3. Ou use as variáveis CSS em qualquer lugar -->
<style>
  .meu-elemento {
    color: var(--primary);
    background: var(--bg);
    border-radius: var(--radius-lg);
    padding: var(--sp-2);
  }
</style>
